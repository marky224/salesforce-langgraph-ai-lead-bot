# Eval-suite-only dependencies — NOT installed into the production image and NOT
# part of the default test job's install. The Dockerfile copies only app/ and
# installs requirements.txt; CI's hermetic `pytest tests/` never collects evals/.
#
# Install for local eval work / recording:
#     pip install -r backend/requirements-eval.txt
-r requirements.txt

# Test runner — pinned to match the hermetic suite (requirements-dev.txt) so the
# eval gate runs under the same pytest/asyncio. Required: the eval tests are async
# (@pytest.mark.asyncio) and pytest-recording does not pin pytest, so without these
# a clean install pulls a bare, asyncio-less pytest and every eval test errors.
pytest==8.4.2
pytest-asyncio==0.26.0

# Deterministic real-model replay: record once against live Grok, replay offline
# in CI (--record-mode=none) as the gate. pytest-recording adds the --record-mode
# flag and the @pytest.mark.vcr marker on top of vcrpy.
vcrpy==8.1.1
pytest-recording==0.13.4

# Calibrated LLM-as-judge (PR D) — eval_live only; never the offline gate, never the
# image. openevals wraps the judge (persona adherence + summary faithfulness); the
# judge model is a frontier Claude (claude-sonnet-4-6) built via app.config.get_llm,
# so langchain-anthropic is required here. Pinned to the 0.3.x line on purpose: the
# 1.x line demands langchain-core>=1.4.6, which would force-upgrade the app's pinned
# 0.3.86 and break langgraph 0.6.x / langchain-openai 0.3.x. Verified with
# `uv pip compile` — these resolve with every app pin unchanged.
openevals==0.2.0
langchain-anthropic==0.3.22
